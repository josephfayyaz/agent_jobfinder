## Part 2: New endpoint — GET /health/ai-status

In `job_agent/routers/system.py`, add after the existing `/health` endpoint:

```python
@router.get("/health/ai-status")
def health_ai_status() -> dict:
    """Return whether a working AI provider is configured."""
    from job_agent.config import settings
    from job_agent.llm import LLMRouter, ProviderUnavailableError

    provider_mode = getattr(settings, "llm_provider_mode", "openai").strip().lower()

    if provider_mode == "openai":
        configured = bool(settings.openai_api_key.strip())
    elif provider_mode == "anthropic":
        configured = bool(settings.anthropic_api_key.strip())
    elif provider_mode in {"local", "lm_studio"}:
        configured = bool(settings.lm_studio_small_model.strip())
    elif provider_mode == "ollama":
        configured = bool(getattr(settings, "ollama_base_url", "").strip())
    elif provider_mode == "mixed":
        configured = bool(
            settings.openai_api_key.strip()
            or settings.anthropic_api_key.strip()
            or settings.lm_studio_small_model.strip()
        )
    else:
        configured = False

    return {"configured": configured, "provider_mode": provider_mode}
```

### 4d. Add JavaScript for provider validation

Add the following JavaScript in the `<script>` block at the bottom of
`onboarding.html` (or in `frontend/static/onboarding.js` if it exists):

```javascript
// Show/hide provider-specific fields based on selected radio
document.querySelectorAll('input[name="provider"]').forEach(radio => {
  radio.addEventListener('change', () => {
    document.getElementById('onboard-openai-fields').classList.toggle('hidden', radio.value !== 'openai');
    document.getElementById('onboard-lmstudio-fields').classList.toggle('hidden', radio.value !== 'local');
    document.getElementById('onboard-ollama-fields').classList.toggle('hidden', radio.value !== 'ollama');
  });
});

async function validateAndSaveProvider() {
  const provider = document.querySelector('input[name="provider"]:checked')?.value || 'openai';
  const statusEl = document.getElementById('provider-validate-status');
  statusEl.textContent = 'Validating…';

  let payload = { llm_provider_mode: provider };
  if (provider === 'openai') {
    const key = document.getElementById('openai-key').value.trim();
    if (!key.startsWith('sk-')) {
      statusEl.textContent = '✗ Enter a valid OpenAI API key (starts with sk-)';
      return;
    }
    payload.openai_api_key = key;
  } else if (provider === 'local') {
    payload.lm_studio_base_url = document.getElementById('lmstudio-url').value.trim();
  } else if (provider === 'ollama') {
    payload.ollama_base_url = document.getElementById('ollama-url').value.trim();
  }

  try {
    const resp = await fetch('/settings/save', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });
    if (!resp.ok) throw new Error(await resp.text());

    // Verify connectivity
    const status = await fetch('/health/ai-status').then(r => r.json());
    if (!status.configured) {
      statusEl.textContent = '✗ Could not verify provider. Check your key or URL.';
      return;
    }

    statusEl.textContent = '✓ Connected';
    // Advance to step 2
    setTimeout(() => goToStep(2), 600);
  } catch (err) {
    statusEl.textContent = '✗ Error: ' + err.message;
  }
}
```
