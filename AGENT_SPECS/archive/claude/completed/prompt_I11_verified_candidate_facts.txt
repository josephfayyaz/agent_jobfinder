# Phase I11 — Verified Candidate Facts (Profile Memory Layer)

## Goal

Add a persistent "profile memory" layer so user-confirmed answers from the apply
flow accumulate over time and automatically strengthen future CV drafts — without
ever touching the curated fact sheet or MasterProfile directly.

Design decision: Option B (separate `VerifiedCandidateFact` table). The saved
answer may not always be a skill — it could be a metric, tool, project detail,
responsibility, achievement, language, or experience evidence. The existing fact
sheet remains the clean source of truth; verified facts form a trusted evidence
layer the AI loads alongside it.

User flow:
1. User goes through `/ui/apply` → answers clarifying questions → draft generated.
2. Draft detail page shows: "Answers you provided for this application — Save as
   verified profile memory?"
3. User clicks Save → stored as `VerifiedCandidateFact`.
4. All future calls to `_generate_draft()` load active facts → writer receives a
   `══ VERIFIED CANDIDATE FACTS ══` prompt block that grows richer over time.
5. `/ui/verified-facts` management page: review, deactivate any fact.

```python
class VerifiedCandidateFact(Base):
    """Persistent profile memory: user-confirmed facts from the apply flow."""
    __tablename__ = "verified_candidate_facts"

    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    candidate_id: Mapped[str | None] = mapped_column(
        String(36),
        ForeignKey("candidates.id"),
        nullable=True,
        index=True,
    )
    gap_label: Mapped[str] = mapped_column(String(200), nullable=False)
    evidence: Mapped[str] = mapped_column(Text, nullable=False)
    fact_type: Mapped[str] = mapped_column(String(50), default="general", nullable=False)
    source_draft_id: Mapped[uuid.UUID | None] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("resume_drafts.id", ondelete="SET NULL"),
        nullable=True,
        index=True,
    )
    source_job_id: Mapped[uuid.UUID | None] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("job_posts.id", ondelete="SET NULL"),
        nullable=True,
        index=True,
    )
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        default=lambda: datetime.now(timezone.utc),
    )
    is_active: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)
    notes: Mapped[str | None] = mapped_column(Text, nullable=True)
```

Alembic, no manual DDL, no `_safe_add_column()` needed here.

```python
@app.post("/verified-facts", status_code=201)
def create_verified_fact(body: dict = Body(...), db: Session = Depends(get_db)):
    from job_agent.models import VerifiedCandidateFact
    import uuid as uuid_mod

    gap_label = str(body.get("gap_label") or "").strip()
    evidence = str(body.get("evidence") or "").strip()
    if not gap_label or not evidence:
        raise HTTPException(status_code=422, detail="gap_label and evidence are required.")

    source_draft_id = None
    raw_draft_id = body.get("source_draft_id")
    if raw_draft_id:
        try:
            source_draft_id = uuid_mod.UUID(str(raw_draft_id))
        except ValueError:
            pass

    source_job_id = None
    raw_job_id = body.get("source_job_id")
    if raw_job_id:
        try:
            source_job_id = uuid_mod.UUID(str(raw_job_id))
        except ValueError:
            pass

    candidate_id = settings.default_candidate_id.strip() or None

    fact = VerifiedCandidateFact(
        candidate_id=candidate_id,
        gap_label=gap_label,
        evidence=evidence,
        fact_type=str(body.get("fact_type") or "general"),
        source_draft_id=source_draft_id,
        source_job_id=source_job_id,
        notes=str(body.get("notes") or "") or None,
    )
    db.add(fact)
    db.commit()
    db.refresh(fact)
    return {
        "id": str(fact.id),
        "gap_label": fact.gap_label,
        "evidence": fact.evidence,
        "fact_type": fact.fact_type,
        "source_draft_id": str(fact.source_draft_id) if fact.source_draft_id else None,
        "created_at": fact.created_at.isoformat(),
        "is_active": fact.is_active,
    }


@app.get("/verified-facts")
def list_verified_facts(db: Session = Depends(get_db)):
    from job_agent.models import VerifiedCandidateFact

    candidate_id = settings.default_candidate_id.strip() or None
    facts = (
        db.query(VerifiedCandidateFact)
        .filter(
            VerifiedCandidateFact.is_active == True,
            VerifiedCandidateFact.candidate_id == candidate_id,
        )
        .order_by(VerifiedCandidateFact.created_at.desc())
        .all()
    )
    return [
        {
            "id": str(f.id),
            "gap_label": f.gap_label,
            "evidence": f.evidence,
            "fact_type": f.fact_type,
            "source_draft_id": str(f.source_draft_id) if f.source_draft_id else None,
            "created_at": f.created_at.isoformat(),
        }
        for f in facts
    ]


@app.delete("/verified-facts/{fact_id}", status_code=204)
def deactivate_verified_fact(fact_id: UUID, db: Session = Depends(get_db)):
    from job_agent.models import VerifiedCandidateFact

    fact = db.get(VerifiedCandidateFact, fact_id)
    if fact is None:
        raise HTTPException(status_code=404, detail="Fact not found.")
    fact.is_active = False
    db.commit()
```

```python
@app.get("/ui/drafts/{draft_id}", include_in_schema=False)
def ui_draft_detail(draft_id: UUID, request: Request, db: Session = Depends(get_db)):
    from job_agent.models import VerifiedCandidateFact

    draft = db.get(ResumeDraft, draft_id)
    if draft is None:
        raise HTTPException(status_code=404, detail="Draft not found.")
    job = db.get(JobPost, draft.job_post_id)
    track = db.get(ResumeTrack, draft.resume_track_id)
    feedback = draft.review_feedback or {}

    # Facts the user confirmed during this specific apply session
    confirmed_facts = feedback.get("confirmed_facts") or []

    # Which of those have already been promoted to profile memory from this draft
    promoted_gaps: set[str] = set()
    if confirmed_facts:
        promoted_gaps = {
            f.gap_label
            for f in db.query(VerifiedCandidateFact)
            .filter(
                VerifiedCandidateFact.source_draft_id == draft_id,
                VerifiedCandidateFact.is_active == True,
            )
            .all()
        }

    return templates.TemplateResponse(
        request=request,
        name="draft_detail.html",
        context=_ui_context(
            request,
            "Draft Detail",
            draft=draft,
            job=job,
            track=track,
            content=draft.content or {},
            review_feedback=feedback,
            compliance_matrix=feedback.get("compliance_matrix"),
            ranked_keywords=feedback.get("ranked_keywords") or [],
            confirmed_facts=confirmed_facts,       # ← new
            promoted_gaps=promoted_gaps,           # ← new
        ),
    )
```

```javascript
async function saveToProfileMemory(btn, gapLabel, evidence, draftId) {
  btn.disabled = true;
  btn.textContent = "Saving…";
  try {
    const resp = await fetch("/verified-facts", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({gap_label: gapLabel, evidence, source_draft_id: draftId}),
    });
    if (resp.ok) {
      btn.textContent = "Saved ✓";
      btn.className = btn.className.replace("border-indigo-300 text-indigo-600", "border-green-300 text-green-700");
      btn.classList.add("cursor-default");
    } else {
      btn.textContent = "Error — try again";
      btn.disabled = false;
    }
  } catch {
    btn.textContent = "Error — try again";
    btn.disabled = false;
  }
}
```

```python
@app.get("/ui/verified-facts", include_in_schema=False)
def ui_verified_facts(request: Request, db: Session = Depends(get_db)):
```

- TemplateResponse keyword args only: `TemplateResponse(request=request, name=..., context=...)`.
- Catch `(ProviderUnavailableError, ValueError)` on all LLM calls (none added here).
- No shared DB sessions across threads.
- Never modify `MasterProfile.global_skills`, `MasterProfile.experiences`, or any `ExperienceFact` / `ProjectFact` row — the curated fact sheet remains untouched.
