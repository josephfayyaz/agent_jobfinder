PHASE I5 — Gmail Setup Wizard (Web UI)
=======================================

## Goal

Gmail currently works via IMAP + App Password — but there is no UI to configure it.
Users must edit `.env` manually and know what `GMAIL_APP_PASSWORD` means.

This phase adds a Gmail setup wizard to the Settings page that:
1. Guides the user through enabling Gmail App Passwords (link + instructions)
2. Lets them input email + app password via the web UI
3. Tests the IMAP connection before saving
4. Shows current Gmail sync status on the Jobs page

## Backend

### `POST /gmail/test-connection`  (NEW)

Test IMAP connection without saving credentials:

```python
@app.post("/gmail/test-connection")
def test_gmail_connection(
    email: str,
    app_password: str,
) -> dict[str, object]:
    """Test IMAP credentials. Returns {ok: bool, error: str|None}."""
    import imaplib
    try:
        conn = imaplib.IMAP4_SSL("imap.gmail.com")
        conn.login(email.strip(), app_password.strip())
        conn.logout()
        return {"ok": True, "error": None}
    except imaplib.IMAP4.error as exc:
        return {"ok": False, "error": str(exc)}
    except Exception as exc:
        return {"ok": False, "error": f"Connection failed: {exc}"}
```

### Add `testGmail()` to settings.html JavaScript:

```javascript
async function testGmail() {
  const form = document.getElementById("gmail-form");
  const email = form.elements["gmail_email"].value.trim();
  const password = form.elements["gmail_app_password"].value.trim();
  const status = document.getElementById("gmail-status");

  if (!email || !password) {
    status.textContent = "Enter email and app password first.";
    return;
  }

  status.textContent = "Testing…";
  const params = new URLSearchParams({ email, app_password: password });
  const resp = await fetch("/gmail/test-connection?" + params.toString(), { method: "POST" });
  const data = await resp.json();
  status.textContent = data.ok ? "✓ Connection successful" : "✗ " + (data.error || "Failed");
}
```

## Tests (`tests/test_gmail_setup.py`)

```python
def test_gmail_status_unconfigured(client, monkeypatch):
    import job_agent.config as cfg
    object.__setattr__(cfg.settings, "gmail_email", "")
    object.__setattr__(cfg.settings, "gmail_app_password", "")
    resp = client.get("/gmail/status")
    assert resp.status_code == 200
    assert resp.json()["configured"] is False

def test_gmail_test_connection_bad_credentials(client, monkeypatch):
    import imaplib
    monkeypatch.setattr(imaplib.IMAP4_SSL, "__init__", lambda *a, **kw: None)
    monkeypatch.setattr(imaplib.IMAP4_SSL, "login",
                        lambda *a, **kw: (_ for _ in ()).throw(imaplib.IMAP4.error("bad credentials")))
    resp = client.post("/gmail/test-connection?email=test@gmail.com&app_password=badpass")
    assert resp.status_code == 200
    data = resp.json()
    assert data["ok"] is False
    assert "bad credentials" in data["error"]

def test_ui_settings_shows_gmail_section(client):
    resp = client.get("/ui/settings")
    assert resp.status_code == 200
    assert "Gmail Integration" in resp.text
    assert "App Password" in resp.text
```
