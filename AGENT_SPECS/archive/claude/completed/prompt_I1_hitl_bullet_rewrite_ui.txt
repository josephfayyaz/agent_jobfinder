PHASE I1 — Human-in-the-Loop Bullet Rewrite Web UI
====================================================

## Goal

The backend for bullet clarify already exists (F1 phase): `BulletClarifyService`,
`BulletClarification` model, `POST /drafts/{id}/clarify`. It is NOT exposed in the web UI.

This phase adds an interactive bullet rewrite flow to `draft_detail.html` so users can
click any experience bullet and get an AI-suggested rewrite — entirely from the browser.

---

## Backend (minimal additions needed)

### `GET /drafts/{draft_id}/clarifications`  (NEW)

Return all existing clarifications for this draft:

```python
@app.get("/drafts/{draft_id}/clarifications")
def list_clarifications(draft_id: str, db: Session = Depends(get_db)) -> list[dict]:
    from job_agent.models import BulletClarification
    rows = db.query(BulletClarification).filter_by(draft_id=draft_id).all()
    return [
        {
            "id": str(r.id),
            "original_bullet": r.original_bullet,
            "suggested_rewrite": r.suggested_rewrite,
            "user_accepted": r.user_accepted,
            "created_at": r.created_at.isoformat() if r.created_at else None,
        }
        for r in rows
    ]
```

### `POST /drafts/{draft_id}/clarify`  (ALREADY EXISTS — verify signature)

Accepts JSON: `{"bullet": "<original text>"}`.
Returns: `{"id": "...", "suggested_rewrite": "..."}`.
If signature differs, adjust the template's fetch call to match.

### `POST /drafts/clarifications/{clarification_id}/accept`  (NEW)

Mark a clarification as accepted (user_accepted = True):

```python
@app.post("/drafts/clarifications/{clar_id}/accept", status_code=204)
def accept_clarification(clar_id: str, db: Session = Depends(get_db)) -> Response:
    from job_agent.models import BulletClarification
    row = db.get(BulletClarification, clar_id)
    if not row:
        raise HTTPException(404)
    row.user_accepted = True
    db.commit()
    return Response(status_code=204)
```

### JavaScript (add to bottom of the template, inside `<script>`):

```javascript
let _currentClarificationId = null;

async function rewriteBullet(btn, draftId, bullet) {
  const panel = document.getElementById("bullet-clarify-panel");
  const loading = document.getElementById("clarify-loading");
  const result = document.getElementById("clarify-result");
  const originalEl = document.getElementById("clarify-original");
  const rewriteEl = document.getElementById("clarify-rewrite");

  originalEl.textContent = bullet;
  loading.classList.remove("hidden");
  result.classList.add("hidden");
  panel.scrollIntoView({ behavior: "smooth", block: "nearest" });

  try {
    const resp = await fetch(`/drafts/${draftId}/clarify`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ bullet }),
    });
    if (!resp.ok) throw new Error(await resp.text());
    const data = await resp.json();
    _currentClarificationId = data.id;
    rewriteEl.textContent = data.suggested_rewrite;
    loading.classList.add("hidden");
    result.classList.remove("hidden");
  } catch (err) {
    loading.classList.add("hidden");
    rewriteEl.textContent = "Error: " + err.message;
    result.classList.remove("hidden");
  }
}

async function acceptRewrite() {
  if (!_currentClarificationId) return;
  await fetch(`/drafts/clarifications/${_currentClarificationId}/accept`, { method: "POST" });
  document.getElementById("clarify-accept-btn").textContent = "✓ Accepted";
  document.getElementById("clarify-accept-btn").disabled = true;
}

function dismissRewrite() {
  document.getElementById("clarify-result").classList.add("hidden");
  _currentClarificationId = null;
}
```

```python
def test_list_clarifications_empty(client, db_session):
    from job_agent.models import ResumeDraft
    import uuid
    draft = ResumeDraft(id=str(uuid.uuid4()), job_post_id=None, track_slug="t",
                        score=80, status="drafted", provider="local")
    db_session.add(draft)
    db_session.commit()
    resp = client.get(f"/drafts/{draft.id}/clarifications")
    assert resp.status_code == 200
    assert resp.json() == []

def test_accept_clarification(client, db_session):
    from job_agent.models import BulletClarification, ResumeDraft
    import uuid
    draft_id = str(uuid.uuid4())
    clar = BulletClarification(
        id=str(uuid.uuid4()),
        draft_id=draft_id,
        original_bullet="Built pipelines.",
        suggested_rewrite="Built ETL pipelines reducing latency by 30%.",
        user_accepted=False,
    )
    db_session.add(clar)
    db_session.commit()
    resp = client.post(f"/drafts/clarifications/{clar.id}/accept")
    assert resp.status_code == 204
    db_session.refresh(clar)
    assert clar.user_accepted is True
```
