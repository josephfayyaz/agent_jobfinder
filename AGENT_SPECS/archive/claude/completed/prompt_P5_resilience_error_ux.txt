# Phase P5 — Resilience and Error UX

## Goal

Add structured error handling across the app:
1. Network-down detection with a top-of-page banner that auto-hides on reconnect.
2. Toast notification system (bottom-right, auto-dismiss) for transient errors.
3. Per-scenario error behaviour: silent retry for scoring, immediate error + Retry
   button for CV drafts, queued auto-retry for background Gmail sync.
4. Model timeout and token-exhaustion handling with clear user-facing messages.
5. Locked action state on job cards with `enrichment_status == "needs_input"`
   (lockout enforced at service level, not just UI).

### 2c. Add retry wrapper for scoring operations

Add a `generate_with_retry()` method to `LLMRouter` for use by fast/scoring
operations only (NOT for draft generation — drafts must fail immediately):

```python
def generate_with_retry(
    self,
    *,
    role: str,
    system_prompt: str,
    user_prompt: str,
    temperature: float = 0.2,
) -> dict[str, Any]:
    """
    Like generate(), but silently retries up to llm_retry_attempts times
    on ProviderUnavailableError before re-raising. Intended for fast
    operations (scoring, analysis) — NOT for draft generation.
    """
    max_attempts = max(1, getattr(settings, "llm_retry_attempts", 2))
    last_exc: Exception | None = None

    for attempt in range(max_attempts):
        try:
            return self.generate(
                role=role,
                system_prompt=system_prompt,
                user_prompt=user_prompt,
                temperature=temperature,
            )
        except (ProviderUnavailableError, ValueError) as exc:
            last_exc = exc
            if attempt < max_attempts - 1:
                import time
                time.sleep(2 ** attempt)  # 1s, 2s backoff
                logger.warning(
                    "LLM call failed (attempt %d/%d), retrying: %s",
                    attempt + 1, max_attempts, exc,
                )

    raise last_exc  # type: ignore[misc]
```

## Part 5: app.js — network detection + toast functions

In `frontend/static/app.js`, add the following (append to the existing file,
do not remove existing code):

```javascript
// ── Network banner ─────────────────────────────────────────────────────────

(function initNetworkMonitor() {
  const banner = document.getElementById('network-banner');
  if (!banner) return;

  let wasOffline = false;

  async function checkConnectivity() {
    try {
      const resp = await fetch('/health/connectivity', { cache: 'no-store' });
      const data = await resp.json();
      if (!data.online) {
        banner.classList.remove('hidden');
        wasOffline = true;
      } else if (wasOffline) {
        banner.classList.add('hidden');
        wasOffline = false;
        showToast('Back online — resuming cloud features.', 'success');
      } else {
        banner.classList.add('hidden');
      }
    } catch {
      // If even the health endpoint fails, assume offline
      banner.classList.remove('hidden');
      wasOffline = true;
    }
  }

  checkConnectivity();
  setInterval(checkConnectivity, 30_000);  // poll every 30s
})();


// ── Toast system ───────────────────────────────────────────────────────────

/**
 * Show a toast notification.
 * @param {string} message   - User-visible message text
 * @param {'info'|'success'|'warning'|'error'} type - Visual style
 * @param {number} duration  - Auto-dismiss delay in ms (0 = no auto-dismiss)
 */
function showToast(message, type = 'info', duration = 6000) {
  const container = document.getElementById('toast-container');
  if (!container) return;

  const colours = {
    info:    'bg-slate-800 text-white',
    success: 'bg-emerald-700 text-white',
    warning: 'bg-amber-500 text-white',
    error:   'bg-red-600 text-white',
  };
  const icons = {
    info: 'ℹ', success: '✓', warning: '⚠', error: '✕'
  };

  const toast = document.createElement('div');
  toast.className = [
    'pointer-events-auto flex items-start gap-3 rounded-xl px-4 py-3 shadow-lg',
    'transform transition-all duration-300 translate-x-full opacity-0',
    colours[type] || colours.info,
  ].join(' ');
  toast.innerHTML = `
    <span class="text-base leading-none mt-0.5">${icons[type] || 'ℹ'}</span>
    <p class="flex-1 text-sm leading-snug">${message}</p>
    <button onclick="this.closest('[data-toast]').remove()"
            class="text-white/70 hover:text-white leading-none text-lg ml-1">×</button>
  `;
  toast.dataset.toast = '1';
  container.appendChild(toast);

  // Animate in
  requestAnimationFrame(() => {
    requestAnimationFrame(() => {
      toast.classList.remove('translate-x-full', 'opacity-0');
    });
  });

  if (duration > 0) {
    setTimeout(() => {
      toast.classList.add('opacity-0');
      setTimeout(() => toast.remove(), 300);
    }, duration);
  }

  return toast;
}

// Expose globally
window.showToast = showToast;
```

```javascript
// In the draft generation JS (wherever generateDraft() or similar function exists):
async function generateDraft(payload) {
  const errorEl = document.getElementById('draft-error');
  const retryBtn = document.getElementById('draft-retry-btn');
  if (errorEl) errorEl.classList.add('hidden');

  try {
    const resp = await fetch('/drafts', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });

    if (!resp.ok) {
      const data = await resp.json().catch(() => ({}));
      const msg = data.detail || 'Draft generation failed. Please retry.';

      if (errorEl) {
        errorEl.textContent = msg;
        errorEl.classList.remove('hidden');
      }
      if (retryBtn) {
        retryBtn.classList.remove('hidden');
        retryBtn.onclick = () => generateDraft(payload);
      }

      showToast(msg, 'error', 0);  // persistent — user must dismiss
      return null;
    }

    return await resp.json();
  } catch (err) {
    showToast('Network error — check your connection.', 'error', 0);
    if (retryBtn) {
      retryBtn.classList.remove('hidden');
      retryBtn.onclick = () => generateDraft(payload);
    }
    return null;
  }
}
```

```python
from __future__ import annotations

from unittest.mock import patch


def test_connectivity_endpoint_online(client):
    """Returns online=True when DNS lookup succeeds."""
    import socket
    with patch("socket.getaddrinfo", return_value=[("", "", "", "", ("8.8.8.8", 443))]):
        resp = client.get("/health/connectivity")
    assert resp.status_code == 200
    assert resp.json()["online"] is True


def test_connectivity_endpoint_offline(client):
    """Returns online=False when DNS lookup fails."""
    import socket
    with patch("socket.getaddrinfo", side_effect=OSError("unreachable")):
        resp = client.get("/health/connectivity")
    assert resp.status_code == 200
    assert resp.json()["online"] is False


def test_draft_generation_blocked_for_empty_description(client, db_session):
    """POST /drafts must return 422 when job description is empty."""

    import uuid

    job = JobPost(
        title="Engineer",
        company="Acme",
        description="",
        status="new",
        enrichment_status="needs_input",
    )
    track = ResumeTrack(title="Engineering", slug="engineering", active=True)
    db_session.add_all([job, track])
    db_session.commit()

    resp = client.post("/drafts", json={
        "job_post_id": str(job.id),
        "resume_track_slug": track.slug,

    assert resp.status_code == 422
    assert "description" in resp.json()["detail"].lower()


def test_llm_retry_wrapper_retries_on_failure(monkeypatch):
    """generate_with_retry() retries up to llm_retry_attempts times."""
    from job_agent.config import settings
    from job_agent.llm import LLMRouter, ProviderUnavailableError

    monkeypatch.setattr(settings, "llm_retry_attempts", 2)
    monkeypatch.setattr(settings, "llm_provider_mode", "openai")
    monkeypatch.setattr(settings, "openai_api_key", "sk-test")

    router = LLMRouter()
    call_count = 0

    def flaky_generate(**kwargs):
        nonlocal call_count
        call_count += 1
        if call_count < 2:
            raise ProviderUnavailableError("transient failure")
        return {"result": "ok"}

    with patch.object(router, "generate", side_effect=flaky_generate):
        result = router.generate_with_retry(
            role="small", system_prompt="s", user_prompt="u"
        )

    assert result == {"result": "ok"}
    assert call_count == 2


def test_llm_retry_wrapper_raises_after_all_attempts(monkeypatch):
```
