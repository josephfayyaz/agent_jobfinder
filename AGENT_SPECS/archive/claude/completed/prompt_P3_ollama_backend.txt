# Phase P3 — Ollama Backend

## Goal

Add Ollama as a third local LLM backend alongside LM Studio. Ollama exposes an
OpenAI-compatible API at `http://localhost:11434/v1/`, so the existing
`_call_lm_studio()` code path is reused with a different base URL. Add Ollama-
specific config fields, a `GET /ollama/models` discovery endpoint, update the
settings UI with an Ollama section, and handle the migration of the old
`llm_provider_mode="local"` value to `"lm_studio"`.

### 2b. Update `generate()` to handle "ollama" and "lm_studio" modes

In the `generate()` method, find the block that checks `provider_mode`. Update
it so:

- `"lm_studio"` and `"local"` both route to `_generate_local()`
  (`"local"` is the legacy value — must be handled for backward compat)
- `"ollama"` routes to a new `_generate_ollama()` method
- `"mixed"` tries `_generate_local()` first, then `_generate_ollama()`, then
  OpenAI/Anthropic cloud

Replace the existing provider_mode dispatch block with:

```python
provider_mode = getattr(settings, "llm_provider_mode", "openai").strip().lower()

# Normalize legacy "local" → "lm_studio"
if provider_mode == "local":
    provider_mode = "lm_studio"

if provider_mode == "openai":
    return self.generate_with_openai(
        model=self._openai_model_for_role(role),
        system_prompt=system_prompt,
        user_prompt=user_prompt,
    )
if provider_mode == "anthropic":
    return self.generate_with_claude(system_prompt=system_prompt, user_prompt=user_prompt)
if provider_mode == "ollama":
    return self._generate_ollama(
        role=role,
        system_prompt=system_prompt,
        user_prompt=user_prompt,
        temperature=temperature,
    )
if provider_mode == "mixed":
    # Try local first, then Ollama, then cloud
    for attempt in ("lm_studio", "ollama"):
        try:
            if attempt == "lm_studio":
                return self._generate_local(
                    role=role, system_prompt=system_prompt,
                    user_prompt=user_prompt, temperature=temperature,
                )
            else:
                return self._generate_ollama(
                    role=role, system_prompt=system_prompt,
                    user_prompt=user_prompt, temperature=temperature,
                )
        except ProviderUnavailableError:
            continue
    # Fall back to cloud
    if self._openai_client is not None:
        return self.generate_with_openai(
            model=self._openai_model_for_role(role),
            system_prompt=system_prompt,
            user_prompt=user_prompt,
        )
    if self._anthropic_client is not None:
        return self.generate_with_claude(
            system_prompt=system_prompt, user_prompt=user_prompt
        )
    raise ProviderUnavailableError(
        "Mixed mode: all providers failed or unconfigured."
    )

# Default: lm_studio
return self._generate_local(
    role=role,
    system_prompt=system_prompt,
    user_prompt=user_prompt,
    temperature=temperature,
)
```

### 2c. Add `_generate_ollama()` method

Add this method to `LLMRouter` after `_generate_local()`:

```python
def _generate_ollama(
    self,
    *,
    role: str,
    system_prompt: str,
    user_prompt: str,
    temperature: float = 0.2,
) -> dict[str, Any]:
    """Route to Ollama's OpenAI-compatible endpoint."""
    model_attr = self._OLLAMA_ROLE_MAP.get(role, "ollama_fast_model")
    model: str = getattr(settings, model_attr, "")

    if not model:
        raise ProviderUnavailableError(
            f"Ollama model for role '{role}' is not configured. "
            f"Set {model_attr.upper()} in your settings."
        )

    # Ollama's OpenAI-compatible endpoint is at {base_url}/v1/chat/completions
    ollama_base = getattr(settings, "ollama_base_url", "http://localhost:11434").rstrip("/")
    ollama_v1_url = f"{ollama_base}/v1"

    # Reuse _call_lm_studio() with Ollama's base URL — same protocol
    # Temporarily override the call with Ollama's URL
    payload = {
        "model": model,
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt},
        ],
        "temperature": temperature,
        "max_tokens": settings.lm_studio_max_tokens_small,  # reuse budget
    }
    try:
        with httpx.Client(timeout=settings.lm_studio_request_timeout_seconds) as client:
            response = client.post(
                f"{ollama_v1_url}/chat/completions",
                json=payload,
            )
            response.raise_for_status()

        body = response.json()
        content: str = body["choices"][0]["message"].get("content") or ""
        if not content.strip():
            raise ProviderUnavailableError(
                f"Ollama model '{model}' returned empty content."
            )
        return self._extract_json(content)

    except ProviderUnavailableError:
        raise
    except Exception as exc:
        raise ProviderUnavailableError(
            f"Ollama request failed for model '{model}': {exc}"
        ) from exc
```

```python
def _ollama_reachable(self) -> bool:
    try:
        ollama_base = getattr(settings, "ollama_base_url", "").rstrip("/")
        if not ollama_base:
            return False
        with httpx.Client(timeout=5.0) as client:
            response = client.get(f"{ollama_base}/api/tags")
            response.raise_for_status()
        return True
    except Exception:
        return False
```

## Part 3: system.py — GET /ollama/models endpoint

In `job_agent/routers/system.py`, add a new endpoint after the existing
`/lm-studio/models` endpoint. Follow the same pattern:

```python
@router.get("/ollama/models")
def get_ollama_models() -> dict:
    """Fetch list of installed models from the local Ollama server."""
    from job_agent.config import settings
    import httpx

    ollama_base = getattr(settings, "ollama_base_url", "http://localhost:11434").rstrip("/")
    try:
        with httpx.Client(timeout=8.0) as client:
            resp = client.get(f"{ollama_base}/api/tags")
            resp.raise_for_status()
        data = resp.json()
        models = [m["name"] for m in data.get("models", [])]
        return {"models": models, "reachable": True}
    except Exception as exc:
        return {"models": [], "reachable": False, "error": str(exc)}
```

Add JavaScript to fetch and populate Ollama model dropdowns:

```javascript
async function fetchOllamaModels() {
  const statusEl = document.getElementById('ollama-models-status');
  statusEl.textContent = 'Fetching…';
  try {
    const resp = await fetch('/ollama/models');
    const data = await resp.json();
    if (!data.reachable || !data.models.length) {
      statusEl.textContent = data.error
        ? '✗ ' + data.error
        : '✗ Ollama not reachable or no models installed.';
      return;
    }
    ['ollama-fast-select', 'ollama-writer-select'].forEach(selectId => {
      const sel = document.getElementById(selectId);
      if (!sel) return;
      const current = sel.value;
      sel.innerHTML = data.models.map(m =>
        `<option value="${m}" ${m === current ? 'selected' : ''}>${m}</option>`
      ).join('');
    });
    statusEl.textContent = `✓ ${data.models.length} models found`;
  } catch (err) {
    statusEl.textContent = '✗ Error: ' + err.message;
  }
}
```

```python
"""P3 — Ollama backend in LLMRouter."""
from __future__ import annotations

from unittest.mock import MagicMock, patch
import pytest


def test_ollama_models_endpoint_reachable(client, monkeypatch):
    """Returns model list when Ollama is running."""
    import httpx

    mock_response = MagicMock()
    mock_response.json.return_value = {
        "models": [{"name": "llama3:8b"}, {"name": "mistral:7b"}]
    }
    mock_response.raise_for_status = MagicMock()

    with patch("httpx.Client") as mock_client:
        mock_client.return_value.__enter__.return_value.get.return_value = mock_response
        resp = client.get("/ollama/models")

    assert resp.status_code == 200
    data = resp.json()
    assert data["reachable"] is True
    assert "llama3:8b" in data["models"]


def test_ollama_models_endpoint_unreachable(client):
    """Returns reachable=False when Ollama is not running."""
    import httpx

    with patch("httpx.Client") as mock_client:
        mock_client.return_value.__enter__.return_value.get.side_effect = (
            httpx.ConnectError("Connection refused")
        )
        resp = client.get("/ollama/models")

    assert resp.status_code == 200
    assert resp.json()["reachable"] is False


def test_llm_router_routes_to_ollama(monkeypatch):
    """generate() calls Ollama endpoint when provider_mode='ollama'."""
    from job_agent.config import settings
    from job_agent.llm import LLMRouter

    monkeypatch.setattr(settings, "llm_provider_mode", "ollama")
    monkeypatch.setattr(settings, "ollama_base_url", "http://localhost:11434")
    monkeypatch.setattr(settings, "ollama_fast_model", "llama3:8b")

    router = LLMRouter()

    with patch.object(router, "_generate_ollama", return_value={"result": "ok"}) as mock_ollama:
        result = router.generate(
            role="small",
            system_prompt="sys",
            user_prompt="user",
```
