PHASE I6 — First-Run Package Checker
======================================

## Goal

When the app starts for the first time on a new machine, the user may be missing
Python packages. This phase adds a preflight check that:

1. Detects missing packages at startup.
2. Attempts to install them with `pip install -e .` (non-blocking, shows progress).
3. If offline or install fails → shows a clear error and stops gracefully.
4. Never crashes silently — every failure has a human-readable explanation.

---

## File: `job_agent/preflight.py`  (NEW)

```python
"""Package availability check and auto-install on first run."""
from __future__ import annotations

import importlib
import subprocess
import sys
from pathlib import Path

from rich.console import Console

console = Console()

REQUIRED_PACKAGES = [
    ("fastapi", "fastapi"),
    ("uvicorn", "uvicorn"),
    ("sqlalchemy", "sqlalchemy"),
    ("httpx", "httpx"),
    ("click", "click"),
    ("rich", "rich"),
    ("pydantic", "pydantic"),
    ("pydantic_settings", "pydantic-settings"),
    ("yaml", "pyyaml"),
    ("questionary", "questionary"),
    ("bs4", "beautifulsoup4"),
    ("jinja2", "jinja2"),
]


def _is_importable(module_name: str) -> bool:
    try:
        importlib.import_module(module_name)
        return True
    except ImportError:
        return False


def _check_network() -> bool:
    """Return True if PyPI is reachable."""
    try:
        import httpx
        resp = httpx.get("https://pypi.org", timeout=5)
        return resp.is_success
    except Exception:
        return False


def check_and_install_packages(project_root: Path | None = None) -> None:
    """
    Check required packages. If any are missing:
    - Try to install with pip install -e .
    - If offline or install fails, print a clear error and raise SystemExit.
    - If all packages are present, return silently.
    """
    missing = [
        pip_name
        for module_name, pip_name in REQUIRED_PACKAGES
        if not _is_importable(module_name)
    ]

    if not missing:
        return  # All good — fast path, no output

    console.print(f"\n[yellow]⚠ Missing packages detected:[/yellow] {', '.join(missing)}")
    console.print("[dim]Attempting automatic installation…[/dim]\n")

    if not _check_network():
        console.print(
            "[red]✗ Cannot reach PyPI (network offline or blocked).[/red]\n"
            "Please connect to the internet and run:\n"
            f"    [bold]pip install -e .[/bold]\n"
            "Then restart the app."
        )
        raise SystemExit(1)

    pyproject = (project_root or Path(".")) / "pyproject.toml"
    if not pyproject.exists():
        console.print(
            "[red]✗ pyproject.toml not found.[/red] Cannot auto-install.\n"
            f"Expected at: {pyproject}\n"
            "Run [bold]pip install -e .[/bold] manually from the project root."
        )
        raise SystemExit(1)

    console.print(f"[dim]Running: pip install -e {project_root or '.'}[/dim]")
    result = subprocess.run(
        [sys.executable, "-m", "pip", "install", "-e", str(project_root or ".")],
        capture_output=True,
        text=True,
    )

    if result.returncode != 0:
        console.print(
            f"[red]✗ Installation failed:[/red]\n{result.stderr[-800:] or result.stdout[-800:]}\n"
            "Fix the error above, then run [bold]pip install -e .[/bold] manually."
        )
        raise SystemExit(1)

    console.print("[green]✓ Packages installed successfully.[/green]")
    console.print("[dim]Continuing startup…[/dim]\n")
```

## Integration Points

### `job_agent/cli_start.py` — call before launch

```python
from job_agent.preflight import check_and_install_packages
check_and_install_packages()
```

### `job_agent/onboard.py` — call at the start of `run_onboarding()`

```python
from job_agent.preflight import check_and_install_packages

```

### `run.sh` — add a pip check step

```bash
#!/usr/bin/env bash
set -e

if [[ "${CONDA_DEFAULT_ENV:-}" != "ML" ]]; then
  eval "$(conda shell.bash hook)"
  conda activate ML
fi

# Quick package preflight (no-op if all packages present)
python -c "from job_agent.preflight import check_and_install_packages; check_and_install_packages()" || exit 1

python -m job_agent.cli start "$@"
```

---

## Important: Never Block on Install Failure

The contract:
- If all packages present → zero output, immediate return.
- If packages missing AND network available → install, then continue.
- If packages missing AND network offline → print error, exit 1.
- If install fails → print error with stderr, exit 1.
- Never catch import errors silently and continue — a half-broken install is worse than a clear error.

## Tests (`tests/test_preflight.py`)

```python
from job_agent.preflight import _is_importable, check_and_install_packages

def test_is_importable_known_package():
    assert _is_importable("sys") is True

def test_is_importable_fake_package():
    assert _is_importable("_this_does_not_exist_xyz_abc") is False

def test_check_passes_when_all_packages_present(monkeypatch):
    """When all packages are importable, check_and_install_packages returns silently."""
    monkeypatch.setattr("job_agent.preflight.REQUIRED_PACKAGES", [("sys", "sys")])
    check_and_install_packages()  # should not raise

def test_check_exits_when_offline_and_missing(monkeypatch, tmp_path):
    """When a package is missing and network is offline, exits with code 1."""
    import pytest
    monkeypatch.setattr(
        "job_agent.preflight.REQUIRED_PACKAGES",
        [("_nonexistent_module_xyz", "nonexistent-package")],
    )
    monkeypatch.setattr("job_agent.preflight._check_network", lambda: False)
    (tmp_path / "pyproject.toml").write_text("[project]\nname='x'\n")
    with pytest.raises(SystemExit) as exc_info:
        check_and_install_packages(project_root=tmp_path)
    assert exc_info.value.code == 1
```
