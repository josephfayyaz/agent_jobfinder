PHASE I8 — Market Skill Gap Dashboard
======================================

## Goal

Add a "Skill Gaps" section to `/ui/analytics` that shows which skills are most
frequently missing from applied-job drafts. This tells the user exactly what to
study or add to their fact sheet to improve hit rate across the roles they target.

Source data: `resume_drafts.review_feedback["missing_keywords"]` — computed in
`CVWorkflowService._generate_draft()` as part of this phase (Part 0 below).

The dashboard aggregates `missing_keywords` across all drafts where
`application_status` is not `"archived"`.

---

## Part 0: Compute `missing_keywords` in `job_agent/services/workflows.py`

### Why it doesn't exist yet

F2 stores `ranked_keywords` (JD requirements ranked by priority) in
`review_feedback["ranked_keywords"]`. The candidate's known skills live in
`profile_dict["global_skills"]` (a list of strings). The comparison was never
computed — `missing_keywords` is absent from every draft in the database.

### Change to `_generate_draft()`

After the keyword extraction block that sets `ranked_keywords`, add this block
immediately before the writer LLM call:

```python
        # Compute missing_keywords = JD requirements absent from candidate's known skills
        missing_keywords: list[str] = []
        if ranked_keywords:
            candidate_skills_lower = {
                s.lower() for s in (profile_dict.get("global_skills") or [])
            }
            missing_keywords = [
                kw["term"]
                for kw in ranked_keywords
                if kw.get("term", "").lower() not in candidate_skills_lower
            ]
```

After the draft is generated and `review` dict is built (i.e., where
`ranked_keywords` is already stored into `review`), also store `missing_keywords`:

```python
        if ranked_keywords:
            review["ranked_keywords"] = ranked_keywords
        if missing_keywords:
            review["missing_keywords"] = missing_keywords
```

This runs in `run_cv_workflow()`, not in a separate service call, so it costs
zero extra LLM calls.

## Part A: New Service — `job_agent/services/skill_gap_analytics.py`

```python
"""Aggregate missing ATS keywords across applied-job drafts."""
from __future__ import annotations

from collections import Counter
from typing import Any

from sqlalchemy.orm import Session

from job_agent.models import ResumeDraft


class SkillGapAnalyticsService:
    def __init__(self, session: Session) -> None:
        self.session = session

    def compute(self, candidate_id: str, top_n: int = 20) -> dict[str, Any]:
        """
        Returns top-N missing skills ranked by frequency across all non-archived
        drafts for the given candidate.
        """
        drafts = (
            self.session.query(ResumeDraft)
            .filter(
                ResumeDraft.candidate_id == candidate_id,
                ResumeDraft.application_status != "archived",
                ResumeDraft.review_feedback.isnot(None),
            )
            .all()
        )

        gap_counter: Counter = Counter()
        total_drafts = 0
        for draft in drafts:
            feedback = draft.review_feedback or {}
            missing = feedback.get("missing_keywords", [])
            if isinstance(missing, list):
                gap_counter.update(k.lower() for k in missing if k)
                total_drafts += 1

        top_gaps = [
            {"skill": skill, "count": count, "pct": round(count / total_drafts * 100) if total_drafts else 0}
            for skill, count in gap_counter.most_common(top_n)
        ]

        return {
            "top_gaps": top_gaps,
            "total_drafts_analyzed": total_drafts,
            "unique_missing_skills": len(gap_counter),
        }
```

## Part B: New Endpoint — `job_agent/main.py`

Add after the existing analytics endpoints (near `/analytics/rejection-patterns`):

```python
@app.get("/analytics/skill-gaps")
def analytics_skill_gaps(
    top_n: int = 20,
    db: Session = Depends(get_db),
):
    candidate_id = settings.default_candidate_id
    if not candidate_id:
        return {"top_gaps": [], "total_drafts_analyzed": 0, "unique_missing_skills": 0}
    svc = SkillGapAnalyticsService(session=db)
    return svc.compute(candidate_id=candidate_id, top_n=top_n)
```

Update the endpoint to detect `HX-Request` header and return the partial:

```python
from starlette.requests import Request

@app.get("/analytics/skill-gaps")
def analytics_skill_gaps(
    request: Request,
    top_n: int = 20,
    db: Session = Depends(get_db),
):
    candidate_id = settings.default_candidate_id
    data = {"top_gaps": [], "total_drafts_analyzed": 0, "unique_missing_skills": 0}
    if candidate_id:
        svc = SkillGapAnalyticsService(session=db)
        data = svc.compute(candidate_id=candidate_id, top_n=top_n)

    if request.headers.get("HX-Request"):
        return TemplateResponse(
            request=request,
            name="partials/skill_gaps.html",
            context=data,
        )
    return data
```

## Tests (`tests/test_skill_gap_analytics.py`)

```python
def test_skill_gap_returns_empty_with_no_drafts(db_session):
    from job_agent.services.skill_gap_analytics import SkillGapAnalyticsService
    svc = SkillGapAnalyticsService(session=db_session)
    result = svc.compute(candidate_id="nonexistent", top_n=20)
    assert result["top_gaps"] == []
    assert result["total_drafts_analyzed"] == 0


def test_skill_gap_aggregates_missing_keywords(db_session):
    from job_agent.models import ResumeDraft
    from job_agent.services.skill_gap_analytics import SkillGapAnalyticsService
    import uuid

    cid = str(uuid.uuid4())
    for _ in range(3):
        d = ResumeDraft(
            id=str(uuid.uuid4()),
            candidate_id=cid,
            application_status="pending",
            review_feedback={"missing_keywords": ["dbt", "spark"]},
        )
        db_session.add(d)
    # One draft missing only "dbt"
    d2 = ResumeDraft(
        id=str(uuid.uuid4()),
        candidate_id=cid,
        application_status="pending",
        review_feedback={"missing_keywords": ["dbt"]},
    )
    db_session.add(d2)
    db_session.commit()

    svc = SkillGapAnalyticsService(session=db_session)
    result = svc.compute(candidate_id=cid, top_n=5)
    skills = {g["skill"]: g["count"] for g in result["top_gaps"]}
    assert skills["dbt"] == 4
    assert skills["spark"] == 3
    assert result["total_drafts_analyzed"] == 4


def test_skill_gap_skips_archived_drafts(db_session):
    from job_agent.models import ResumeDraft
    from job_agent.services.skill_gap_analytics import SkillGapAnalyticsService
    import uuid

    cid = str(uuid.uuid4())
    d = ResumeDraft(
        id=str(uuid.uuid4()),
        candidate_id=cid,
        application_status="archived",
        review_feedback={"missing_keywords": ["python"]},
    )
    db_session.add(d)
    db_session.commit()

    svc = SkillGapAnalyticsService(session=db_session)
    result = svc.compute(candidate_id=cid, top_n=5)
    assert result["total_drafts_analyzed"] == 0


def test_analytics_skill_gaps_endpoint(client):
    resp = client.get("/analytics/skill-gaps")
    assert resp.status_code == 200
    data = resp.json()
    assert "top_gaps" in data
    assert "total_drafts_analyzed" in data
```
