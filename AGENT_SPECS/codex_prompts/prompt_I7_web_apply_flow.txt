PHASE I7 — Web Interactive Apply Flow
======================================

## Goal

Port the `start_apply` CLI experience into the web UI so users who don't use the
terminal can still: paste a job description → see the match score → answer up to 3
clarifying questions → have a draft generated.

Currently, `POST /workflow` silently enqueues a CV generation task with no user
input. A non-technical user lands on `/ui/jobs`, clicks "Apply", and gets a draft
that may miss relevant context (the CLI would have asked 2–3 targeted questions).

This phase adds `/ui/apply` — a multi-step page covering the same flow as
`start_apply/cli.py` but driven by HTMX-powered steps, no page reloads.

---

## Flow (mirrors start_apply/cli.py)

```
Step 1  ─── Paste job info / URL  ───►  Step 2: Score + matched/gaps shown
Step 2  ─── Confirm score  ──────────►  Step 3: Clarifying questions (0–3)
Step 3  ─── Answer questions  ───────►  Step 4: "Generating…" spinner
Step 4  ─── Draft ready  ────────────►  Redirect to /ui/drafts/{id}
```

If no gaps (strong match ≥ 85 and 0 gaps), Step 3 is skipped.

## Backend — new endpoints

### `POST /apply/parse` — parse raw job text

Accepts: `application/x-www-form-urlencoded`, field `job_text`.

```python
@app.post("/apply/parse")
async def apply_parse(job_text: str = Form(...), db: Session = Depends(get_db)):
    from start_apply.job_parser import parse_job_info
    from start_apply.scorer import score_match
    from job_agent.services.resource_loader import ResourceLoader

    job = parse_job_info(job_text)
    fact_sheet = ResourceLoader(db).load_fact_sheet_text()   # uses existing helper
    result = score_match(job, fact_sheet)
    return {
        "job": job,
        "score": result["score"],
        "matched": result["matched"],
        "gaps": result["gaps"],
        "summary": result["summary"],
    }
```

### `POST /apply/generate` — create draft from job + confirmed answers

Accepts JSON:
```json
{
  "job": {"title": "...", "company": "...", "location": "...", "description": "..."},
  "answers": [{"gap": "SQL", "answer": "yes, built 3 pipelines", "confirmed": true}],
  "track_slug": "data-engineering-analytics"
}
```

```python
@app.post("/apply/generate")
async def apply_generate(body: dict, db: Session = Depends(get_db)):
    from job_agent.models import JobPost, WorkflowTask
    import uuid
    from datetime import datetime, timezone

    job_data = body["job"]
    # Upsert the job post
    existing = db.query(JobPost).filter_by(
        company=job_data["company"], title=job_data["title"]
    ).first()
    if existing:
        job = existing
    else:
        job = JobPost(
            id=uuid.uuid4(),
            company=job_data.get("company", "Unknown"),
            title=job_data.get("title", "Unknown"),
            description=job_data.get("description", ""),
            location=job_data.get("location", ""),
            source_name="manual",
            canonical_apply_url="",
            created_at=datetime.now(timezone.utc),
            last_seen_at=datetime.now(timezone.utc),
        )
        db.add(job)
        db.commit()
        db.refresh(job)

    # Store confirmed answers in the task payload for the workflow
    task = WorkflowTask(
        kind="cv_workflow",
        status="pending",
        payload={
            "job_post_id": str(job.id),
            "resume_track_slug": body.get("track_slug", ""),
            "confirmed_facts": body.get("answers", []),
        },
    )
    db.add(task)
    db.commit()
    db.refresh(task)
    return {"task_id": str(task.id), "job_id": str(job.id)}
```

### `GET /apply/status/{task_id}` — poll task status (for HTMX polling)

```python
@app.get("/apply/status/{task_id}")
async def apply_status(task_id: str, db: Session = Depends(get_db)):
    from job_agent.models import WorkflowTask, ResumeDraft

    task = db.get(WorkflowTask, task_id)
    if not task:
        raise HTTPException(404)
    draft_id = None
    if task.status == "completed":
        draft = db.query(ResumeDraft).filter_by(
            job_post_id=task.payload.get("job_post_id")
        ).order_by(ResumeDraft.created_at.desc()).first()
        if draft:
            draft_id = str(draft.id)
    return {"status": task.status, "draft_id": draft_id}
```

<script>
let _parsedJob = null;
let _gaps = [];
let _questions = [];
let _taskId = null;
let _pollInterval = null;

async function submitJobText() {
  const text = document.getElementById("job-text-input").value.trim();
  const errEl = document.getElementById("step1-error");
  errEl.classList.add("hidden");
  if (!text) { errEl.textContent = "Please paste some job info first."; errEl.classList.remove("hidden"); return; }

  const btn = event.currentTarget;
  btn.disabled = true;
  btn.textContent = "Parsing…";

  try {
    const resp = await fetch("/apply/parse", {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({ job_text: text }),
    });
    if (!resp.ok) throw new Error(await resp.text());
    const data = await resp.json();
    _parsedJob = data.job;
    _gaps = data.gaps || [];
    showStep2(data);
  } catch (err) {
    errEl.textContent = "Error: " + err.message;
    errEl.classList.remove("hidden");
    btn.disabled = false;
    btn.textContent = "Parse & Score →";
  }
}

function showStep2(data) {
  document.getElementById("step1").classList.add("hidden");
  document.getElementById("step2").classList.remove("hidden");
  document.getElementById("step2-ind").classList.remove("text-slate-400");
  document.getElementById("step2-ind").classList.add("font-semibold", "text-indigo-600");

  const score = data.score ?? "–";
  document.getElementById("score-badge").textContent = score + "/100";
  document.getElementById("score-summary").textContent = data.summary || "";

  const matchedList = document.getElementById("matched-list");
  const gapsList = document.getElementById("gaps-list");
  matchedList.innerHTML = (data.matched || []).slice(0, 6).map(s => `<li>${s}</li>`).join("");
  gapsList.innerHTML = (data.gaps || []).map(s => `<li>${s}</li>`).join("");
}

async function proceedToQuestions() {
  document.getElementById("step2").classList.add("hidden");
  document.getElementById("step3-ind").classList.remove("text-slate-400");
  document.getElementById("step3-ind").classList.add("font-semibold", "text-indigo-600");

  if (_gaps.length === 0) {
    // No gaps → skip to generation
    await generateDraft();
    return;
  }

  // Fetch questions
  const resp = await fetch("/apply/questions", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ job: _parsedJob, gaps: _gaps }),
  });
  const data = await resp.json();
  _questions = data.questions || [];
  renderQuestions();
  document.getElementById("step3").classList.remove("hidden");
}

function renderQuestions() {
  const container = document.getElementById("questions-container");
  container.innerHTML = _questions.map((q, i) => `
    <div class="border border-slate-200 rounded-lg p-4 space-y-2">
      <p class="text-xs font-semibold text-slate-500 uppercase">Gap: ${q.gap}</p>
      <p class="text-sm text-slate-700 font-medium">${q.question}</p>
      <p class="text-xs text-slate-400 italic">${q.why}</p>
      <input type="text" id="answer-${i}"
        class="w-full text-sm border border-slate-300 rounded px-3 py-1.5 focus:ring-2 focus:ring-indigo-400"
        placeholder="Your answer (or leave blank to skip)" />
    </div>
  `).join("");
}

async function generateDraft() {
  document.getElementById("step3").classList.add("hidden");
  document.getElementById("step4").classList.remove("hidden");
  document.getElementById("step4-ind").classList.remove("text-slate-400");
  document.getElementById("step4-ind").classList.add("font-semibold", "text-indigo-600");

  const answers = _questions.map((q, i) => {
    const answer = (document.getElementById(`answer-${i}`)?.value || "").trim();
    return { gap: q.gap, question_id: q.id || `q${i}`, answer, confirmed: !!answer };
  });

  const trackSlug = document.getElementById("track-select")?.value || "";

  const resp = await fetch("/apply/generate", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ job: _parsedJob, answers, track_slug: trackSlug }),
  });
  const data = await resp.json();
  _taskId = data.task_id;
  startPolling();
}

function startPolling() {
  _pollInterval = setInterval(async () => {
    const resp = await fetch(`/apply/status/${_taskId}`);
    const data = await resp.json();
    const statusEl = document.getElementById("status-text");
    statusEl.textContent = `Status: ${data.status}`;
    if (data.status === "completed" && data.draft_id) {
      clearInterval(_pollInterval);
      window.location.href = `/ui/drafts/${data.draft_id}`;
    } else if (data.status === "failed") {
      clearInterval(_pollInterval);
      statusEl.textContent = "Generation failed. Check the worker logs.";
    }
  }, 3000);
}
</script>

@app.get("/ui/apply")
def ui_apply(request: Request, db: Session = Depends(get_db)):
    from job_agent.models import ResumeTrack
    tracks = db.query(ResumeTrack).filter_by(active=True).order_by(ResumeTrack.title).all()
    return TemplateResponse(request=request, name="apply.html", context={"tracks": tracks})

def test_apply_parse_returns_score(client, monkeypatch):
    from start_apply import job_parser, scorer
    monkeypatch.setattr(job_parser, "parse_job_info", lambda text: {"title": "DE", "company": "A", "location": "IT", "description": text})
    monkeypatch.setattr(scorer, "score_match", lambda job, fs: {"score": 80, "matched": ["Python"], "gaps": ["SQL"], "summary": "Good match"})
    monkeypatch.setattr("job_agent.services.resource_loader.ResourceLoader.load_fact_sheet_text", lambda self: "fact sheet text")
    resp = client.post("/apply/parse", data={"job_text": "Data Engineer role at Acme"})
    assert resp.status_code == 200
    data = resp.json()
    assert data["score"] == 80
    assert "Python" in data["matched"]
