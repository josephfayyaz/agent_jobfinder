PHASE I4 — CV Output Folder Selection + Company/Date Naming
=============================================================

## Goal

1. Let users choose where generated CVs and cover letters are saved (instead of hardcoded `generated/`).
2. Name output files as `<Company>_<Date>_<DraftID>.tex/.pdf` so they are easy to find.
3. Expose this setting in onboarding (UX2), CLI settings, and web settings page.

---

## Part B: File Naming Convention

### `job_agent/services/documents.py` — update `render_resume_tex()`

Change the output filename from `{draft_id}.tex` to `{company}_{date}_{draft_id[:8]}.tex`:

```python
import re
from datetime import date

def _safe_filename(text: str) -> str:
    """Convert company name to filesystem-safe string."""
    text = re.sub(r"[^\w\s-]", "", text or "Unknown")
    return re.sub(r"[\s]+", "_", text.strip())[:30]

def render_resume_tex(profile, job, track, content, draft_id: str) -> str:
    artifacts_dir = Path(settings.artifacts_dir)
    artifacts_dir.mkdir(parents=True, exist_ok=True)

    company_slug = _safe_filename(job.company if job else "Unknown")
    date_str = date.today().strftime("%Y-%m-%d")
    filename = f"{company_slug}_{date_str}_{draft_id[:8]}.tex"
    tex_path = artifacts_dir / filename
    ...
```

Apply the same naming to cover letter files in `cover_letter.py`:
```python
filename = f"{company_slug}_{date_str}_{draft_id[:8]}_cover.txt"
```

## Part C: Onboarding Wizard (UX2)

In `job_agent/onboard.py`, add Step 5b (after fact sheet, before summary):

```python
artifacts_default = str(existing_settings.get("artifacts_dir") or settings.artifacts_dir or "generated")
artifacts_input = _prompt_text(
    "Folder to save generated CVs and cover letters?",
    default=artifacts_default,
    non_interactive=non_interactive,
)
if artifacts_input:
    resolved = Path(artifacts_input).expanduser()
    updates["artifacts_dir"] = str(resolved)
    summary["CV Output Folder"] = str(resolved)
```

def test_render_tex_uses_company_name_in_filename(tmp_path, monkeypatch):
    monkeypatch.setattr("job_agent.services.documents.settings.artifacts_dir", tmp_path)
    from job_agent.services.documents import render_resume_tex
    from unittest.mock import MagicMock
    import uuid

    job = MagicMock()
    job.company = "Acme Corp"
    draft_id = str(uuid.uuid4())

    # render_resume_tex needs a minimal profile + content
    profile = MagicMock()
    profile.full_name = "Test User"
    profile.email = "test@example.com"

    content = {
        "professional_summary": "Test",
        "selected_experiences": [],
        "selected_projects": [],
        "selected_skills": [],
    }

    tex_path = render_resume_tex(profile, job, MagicMock(), content, draft_id)
    assert "Acme_Corp" in tex_path or "Acme" in tex_path
    assert draft_id[:8] in tex_path

def test_artifacts_dir_created_if_missing(tmp_path, monkeypatch):

    monkeypatch.setattr("job_agent.services.documents.settings.artifacts_dir", new_dir)
    from job_agent.services.documents import render_resume_tex
    ...  # call render_resume_tex with minimal args

def test_settings_save_accepts_artifacts_dir(client, tmp_path, monkeypatch):
    from job_agent import user_config
